Dark mode (default)

JavaScript

// Function to add two numbers
const add = (a, b) => a + b; /* inline comment */
class Calculator {
	constructor(name = "docuK") { this.name = name; }
	sum(...nums) { return nums.reduce(add, 0); }
}
let total = add(1, 2.5);
console.log(`Total = ${total}`, true, null, undefined, /ab+c/gi);
